const Card = ({ children, className = '', onClick }) => (
  <div onClick={onClick} className={`bg-white rounded-2xl border border-emerald-100 shadow-sm overflow-hidden ${onClick ? 'cursor-pointer active:scale-95 transition-transform' : ''} ${className}`}>
    {children}
  </div>
);

const Button = ({ children, variant = 'primary', className = '', type = 'button', onClick, disabled = false }) => {
  const base = "inline-flex items-center justify-center rounded-xl font-medium transition-colors focus:outline-none focus:ring-2 focus:ring-emerald-500 focus:ring-offset-2 disabled:opacity-50 disabled:pointer-events-none";
  const variants = {
    primary: "bg-emerald-600 text-white hover:bg-emerald-700 active:bg-emerald-800 shadow-sm",
    secondary: "bg-emerald-50 text-emerald-700 hover:bg-emerald-100",
    outline: "border-2 border-emerald-200 text-emerald-700 hover:bg-emerald-50",
    ghost: "text-gray-600 hover:bg-gray-100",
    danger: "bg-red-50 text-red-600 hover:bg-red-100"
  };
  const sizes = "px-4 py-2.5 text-sm";
  return (
    <button type={type} onClick={onClick} disabled={disabled} className={`${base} ${variants[variant]} ${sizes} ${className}`}>
      {children}
    </button>
  );
};

const Input = ({ label, icon: Icon, error, className = '', ...props }) => (
  <div className={`space-y-1.5 ${className}`}>
    {label && <label className="text-sm font-medium text-gray-700">{label}</label>}
    <div className="relative">
      {Icon && <div className="absolute inset-y-0 left-0 pl-3 flex items-center pointer-events-none text-gray-400"><Icon size={18} /></div>}
      <input 
        className={`w-full rounded-xl border ${error ? 'border-red-300 focus:ring-red-500' : 'border-gray-200 focus:border-emerald-500 focus:ring-emerald-500'} bg-gray-50/50 px-4 py-2.5 text-sm transition-all outline-none focus:ring-2 focus:ring-opacity-20 ${Icon ? 'pl-10' : ''}`}
        {...props}
      />
    </div>
    {error && <p className="text-xs text-red-500">{error}</p>}
  </div>
);

const Badge = ({ children, variant = 'default' }) => {
  const variants = {
    default: "bg-gray-100 text-gray-700",
    success: "bg-green-100 text-green-700",
    warning: "bg-yellow-100 text-yellow-700",
    danger: "bg-red-100 text-red-700",
    emerald: "bg-emerald-100 text-emerald-700",
  };
  return <span className={`px-2.5 py-0.5 rounded-full text-xs font-semibold ${variants[variant]}`}>{children}</span>;
};

const ImageWithFallback = ({ src, alt, className = '' }) => {
  const [error, setError] = useState(false);
  if (!src || error) {
    return <div className={`bg-gray-100 flex items-center justify-center text-gray-400 ${className}`}><ImageIcon size={24} /></div>;
  }
  return <img src={src} alt={alt} className={`object-cover ${className}`} onError={() => setError(true)} />;
};

const EntryDataScreen = ({ entries, addEntry }) => {
  const [type, setType] = useState('Ember Tumpuk');
  const [awal, setAwal] = useState('');
  const [pakai, setPakai] = useState('');
  const [showToast, setShowToast] = useState(false);
  
  // Auto calculate
  const sisa = useMemo(() => {
    const wAwal = parseFloat(awal) || 0;
    const wPakai = parseFloat(pakai) || 0;
    return wAwal - wPakai;
  }, [awal, pakai]);

  const isInvalid = sisa < 0 || !awal;

  const handleSubmit = (e) => {
    e.preventDefault();
    if (isInvalid) return;
    addEntry({
      id: Date.now(), date: new Date().toISOString().split('T')[0],
      type, startWeight: parseFloat(awal), usage: parseFloat(pakai)||0, endWeight: sisa, note: ''
    });
    setAwal(''); setPakai('');
    
    // Tampilkan Toast
    setShowToast(true);

const AdminUsers = ({ users, toggleUserStatus, addUser, updateUser, deleteUser }) => {
  const [isModalOpen, setIsModalOpen] = useState(false);
  const [modalMode, setModalMode] = useState('add'); // 'add' | 'edit'
  const [formData, setFormData] = useState({ id: null, name: '', phone: '', location: '' });

  const openModal = (mode, user = null) => {
    setModalMode(mode);
    if (user) {
      setFormData(user);
    } else {
      setFormData({ id: null, name: '', phone: '', location: '' });
    }
    setIsModalOpen(true);
  };

  const handleSubmit = (e) => {
    e.preventDefault();
    if (modalMode === 'add') {
      addUser(formData);
    } else {
      updateUser(formData.id, formData);
    }
    setIsModalOpen(false);
  };

  const handleDelete = (id) => {
    if (window.confirm("Apakah Anda yakin ingin menghapus mitra ini secara permanen?")) {
      deleteUser(id);
    }
  };

  return (
    <div className="flex flex-col h-full bg-slate-50 pb-20 relative">
      <div className="bg-white px-5 py-4 border-b border-slate-200 sticky top-0 z-10 flex justify-between items-center shadow-sm">
        <h1 className="text-xl font-bold text-slate-800">Manajemen Mitra</h1>
        <button onClick={() => openModal('add')} className="bg-slate-800 text-white p-2 rounded-full hover:bg-slate-900 transition-colors shadow-sm">
          <PlusCircle size={20} />
        </button>
      </div>
      <div className="p-5 space-y-3">
        {users.filter(u => u.role !== 'admin').map(u => (
          <Card key={u.id} className="p-4 border-slate-100">
            <div className="flex justify-between items-center mb-3">
              <h3 className="font-bold text-slate-800">{u.name}</h3>
              <Badge variant={u.status === 'active' ? 'success' : 'danger'}>
                {u.status === 'active' ? 'Aktif' : 'Nonaktif'}
              </Badge>
            </div>
            <div className="text-sm text-slate-600 space-y-1 mb-4">
              <p className="flex items-center gap-2"><Phone size={14}/> {u.phone}</p>
              <p className="flex items-center gap-2"><MapPin size={14}/> {u.location || 'Belum diatur'}</p>
            </div>
            <div className="flex justify-between items-center border-t border-slate-100 pt-3">
               <div className="flex gap-2">
                 <button onClick={() => openModal('edit', u)} className="p-1.5 text-slate-500 bg-slate-100 rounded-md hover:bg-indigo-50 hover:text-indigo-600 transition-colors">
                   <Edit size={16} />
                 </button>
                 <button onClick={() => handleDelete(u.id)} className="p-1.5 text-slate-500 bg-slate-100 rounded-md hover:bg-red-50 hover:text-red-600 transition-colors">
                   <Trash2 size={16} />
                 </button>
               </div>
               <Button 
                variant={u.status === 'active' ? 'outline' : 'primary'} 
                className={`!py-1.5 !px-3 !text-xs ${u.status === 'active' ? '!border-red-200 !text-red-600 hover:!bg-red-50' : '!bg-slate-800'}`}
                onClick={() => toggleUserStatus(u.id)}
               >
                 {u.status === 'active' ? 'Nonaktifkan' : 'Aktifkan'}
               </Button>
            </div>
          </Card>
        ))}
      </div>

      {/* Modal Tambah/Edit Mitra */}
      {isModalOpen && (
        <div className="absolute inset-0 z-50 flex items-center justify-center p-4 bg-slate-900/40 backdrop-blur-sm">
          <div className="bg-white w-full max-w-sm rounded-2xl shadow-xl overflow-hidden flex flex-col max-h-full">
            <div className="px-5 py-4 border-b border-slate-100 flex justify-between items-center bg-slate-50 shrink-0">

            </div>

                value={formData.name} onChange={(e) => setFormData({...formData, name: e.target.value})}
              />
              <Input 
                label="No. Handphone" placeholder="0812..." required icon={Phone}
                value={formData.phone} onChange={(e) => setFormData({...formData, phone: e.target.value})}

                label="Lokasi / RT RW" placeholder="RT 01 / RW 02" required icon={MapPin}
                value={formData.location} onChange={(e) => setFormData({...formData, location: e.target.value})}
              />
              <Button type="submit" className="w-full !mt-6 !bg-slate-800 hover:!bg-slate-900">
                {modalMode === 'add' ? 'Simpan Mitra' : 'Perbarui Data'}
              </Button>

  );

// --- MAIN APP COMPONENT ---
export default function App() {
  // Auth & Global State
  const [user, setUser] = useState(null); // null = guest, {...} = logged in
  const [activeTab, setActiveTab] = useState('home'); // for bottom nav (user) or admin nav
  const [currentScreen, setCurrentScreen] = useState('login'); // login, register, app, tutorial, cs
  
  // Mock Data States
  const [users, setUsers] = useState(INITIAL_USERS);
  const [entries, setEntries] = useState(MOCK_ENTRIES);
  const [logs, setLogs] = useState(MOCK_MONITORING);

  // Handlers
  const handleLogin = (userData) => {
    setUser(userData);
    setCurrentScreen('app');
    setActiveTab(userData.role === 'admin' ? 'dashboard' : 'home');
  };

  const handleRegister = (newUserData) => {
    setUsers([...users, newUserData]);
    handleLogin(newUserData);
  };

  const handleLogout = () => {
    setUser(null);
    setCurrentScreen('login');
  };

  const addEntry = (entry) => setEntries([...entries, entry]);
  const addLog = (log) => setLogs([...logs, log]);
  const addLogComment = (id, text) => setLogs(logs.map(l => l.id === id ? { ...l, adminComment: text } : l));
  
  // Manajemen CRUD Admin
  const addUserAdmin = (userData) => {
    setUsers([...users, { id: Date.now(), role: 'user', status: 'active', ...userData }]);
  };
  const updateUserAdmin = (id, userData) => {
    setUsers(users.map(u => u.id === id ? { ...u, ...userData } : u));
  };
  const deleteUserAdmin = (id) => {
    setUsers(users.filter(u => u.id !== id));
  };
  
  const toggleUserStatus = (id) => {
    setUsers(users.map(u => u.id === id ? { ...u, status: u.status === 'active' ? 'inactive' : 'active' } : u));
  };

  // Render logic based on state
  const renderScreen = () => {
    if (currentScreen === 'login') return <LoginScreen onLogin={handleLogin} onNavigate={setCurrentScreen} />;
    if (currentScreen === 'register') return <RegisterScreen onRegister={handleRegister} onNavigate={setCurrentScreen} />;
    if (currentScreen === 'tutorial') return <TutorialScreen onBack={() => setCurrentScreen('app')} />;
    if (currentScreen === 'cs') return <CustomerServiceScreen onBack={() => setCurrentScreen('app')} />;
    
    // Authenticated App Area
    if (user?.role === 'user') {
      return (
        <>
          {activeTab === 'home' && <HomeScreen user={user} setScreen={setCurrentScreen} />}
          {activeTab === 'calendar' && <CalendarScreen />}
          {activeTab === 'entry' && <EntryDataScreen entries={entries.filter(e => e.userId === user.id)} addEntry={addEntry} />}
          {activeTab === 'monitor' && <MonitoringScreen monitoringLogs={logs.filter(l => l.userId === user.id)} addLog={addLog} />}
          <BottomNav activeTab={activeTab} setTab={setActiveTab} role="user" />
          
          {/* Global Logout Button for Mock purposes (Floating) */}
          <button onClick={handleLogout} className="absolute top-4 right-4 z-50 p-2 bg-white/20 backdrop-blur-md rounded-full text-white shadow-sm border border-white/30">
            <LogOut size={18} />
          </button>
        </>
      );
    }

    if (user?.role === 'admin') {
      return (
        <>
          {activeTab === 'dashboard' && <AdminDashboard />}
          {activeTab === 'reports' && <AdminReports logs={logs} addLogComment={addLogComment} />}
          {activeTab === 'users' && (
            <AdminUsers 
              users={users} 
              toggleUserStatus={toggleUserStatus} 
              addUser={addUserAdmin}
              updateUser={updateUserAdmin}
              deleteUser={deleteUserAdmin}
            />
          )}
          <AdminNav activeTab={activeTab} setTab={setActiveTab} />

          <button onClick={handleLogout} className="absolute top-4 right-4 z-50 p-2 bg-slate-900/20 backdrop-blur-md rounded-full text-white shadow-sm border border-white/10">
            <LogOut size={18} />
          </button>
        </>
      );
    }
    
    return null;
  };

  return (
    <MobileFrame>
      {renderScreen()}
    </MobileFrame>
  );
}
